        function executeTriageMatching() {

            // A. Instantiation of parameter objects from document fields

            const targetSeverity = parseInt(document.getElementById('severity').value);

            const targetUrgency = document.getElementById('urgency').value;

            const enforceFree = document.getElementById('cost').value === "true";


            // B. Multi-Factor Weight Allocation Matrix

            const parameterWeights = { severity: 0.50, cost: 0.30, accessibility: 0.20 };

            const weightSum = parameterWeights.severity + parameterWeights.cost + parameterWeights.accessibility;


            // C. Processing pipeline for algorithmic evaluations

            let operationalMatches = resourceDatabase.map(resource => {

                // Safety verification metric (Ensures capability thresholds match required needs)

                let severityVector = (targetSeverity <= resource.maxSeverityHandled) ? 100 : 0;

                

                // Economic compliance scoring framework

                let costVector = 0;

                if (enforceFree && resource.isFree) costVector = 100;

                else if (!enforceFree) costVector = 100;

                else costVector = 20;


                // Temporal resource availability vector calculation

                let availabilityVector = (targetUrgency === "high" && resource.immediateAvailability) ? 100 : (targetUrgency === "high" ? 10 : 100);


                // Algorithmic dot product representation (Weighted average calculation)

                let optimizationMetric = (

                    (severityVector * parameterWeights.severity) +

                    (costVector * parameterWeights.cost) +

                    (availabilityVector * parameterWeights.accessibility)

                ) / weightSum;


                return {

                    ...resource,

                    matchingIndexScore: Math.round(optimizationMetric)

                };

            });
